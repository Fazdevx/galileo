<script>
    import { DEFAULT_DATA } from "../data/olimpiadasStore";
    (async () => {
        // Inline implementation to avoid dynamic imports
        const STORAGE_KEY = 'galileo-olimpiadas-v2';
        
        const onDataChanged = (callback) => {
            const handler = () => callback();
            window.addEventListener('galileo-data-changed', handler);
            window.addEventListener('storage', (e) => {
                if (e.key === STORAGE_KEY) handler();
            });
        };
        
        const calculateStandings = (games, sections) => {
            const stats = sections.map((s) => ({
                id: s.id,
                name: s.name,
                initial: s.initial,
                color: s.color,
                pj: 0,
                g: 0,
                e: 0,
                p: 0,
                pts: 0,
            }));

            games.forEach((g) => {
                if (!g.status.includes('Finalizado')) return;
                const l = stats.find((x) => x.name === g.local);
                const v = stats.find((x) => x.name === g.visit);
                if (!l || !v) return;
                l.pj++;
                v.pj++;
                if (g.localScore > g.visitScore) {
                    l.g++;
                    l.pts += 3;
                    v.p++;
                } else if (g.localScore < g.visitScore) {
                    v.g++;
                    v.pts += 3;
                    l.p++;
                } else {
                    l.e++;
                    v.e++;
                    l.pts += 1;
                    v.pts += 1;
                }
            });

            stats.sort((a, b) => b.pts - a.pts || b.g - a.g);
            return stats;
        };
        
        const DEFAULT_DATA_COPY = JSON.parse(JSON.stringify(DEFAULT_DATA));

        const loadFromFirebase = async () => {
            try {
                const { fetchData } = await import('../data/api');
                const data = await fetchData();
                if (data && data.sections && data.sports && data.games) {
                    return data;
                }
            } catch (e) {
                console.warn('[Olimpiadas] Firebase error:', e);
            }
            return null;
        };

        const loadFromLocal = () => {
            try {
                const raw = localStorage.getItem(STORAGE_KEY);
                if (!raw) return null;
                const parsed = JSON.parse(raw);
                return {
                    sections: Array.isArray(parsed.sections) ? parsed.sections : DEFAULT_DATA.sections,
                    sports: Array.isArray(parsed.sports) ? parsed.sports : DEFAULT_DATA.sports,
                    games: Array.isArray(parsed.games) ? parsed.games : DEFAULT_DATA.games,
                    heroStats: parsed.heroStats || DEFAULT_DATA.heroStats,
                };
            } catch {
                return null;
            }
        };

        const load = async (defaultData) => {
            // Try Firebase first
            const firebaseData = await loadFromFirebase();
            if (firebaseData) {
                console.log('[Olimpiadas] Using Firebase data');
                return firebaseData;
            }
            // Fallback to localStorage
            const localData = loadFromLocal();
            if (localData) {
                console.log('[Olimpiadas] Using local data');
                return localData;
            }
            return defaultData;
        };
        const list = document.querySelector("[data-sections-list]");
        if (!list) return;
        list.innerHTML = sections
    const renderSections = (sections) => {
        const list = document.querySelector("[data-sections-list]");
        if (!list) return;
        list.innerHTML = sections
            .map(
                (s) => `
      <span class="inline-flex items-center gap-2 rounded-full ${s.color} px-3 py-1.5 text-xs font-bold text-white">
        <span class="flex h-5 w-5 items-center justify-center rounded-full bg-white/20 text-[10px]">${s.initial || "?"}</span>
        ${s.name}
      </span>
    `,
            )
            .join("");
    };

    const renderSports = (sports) => {
        const list = document.querySelector("[data-sports-list]");
        if (!list) return;
        list.innerHTML = sports
            .map(
                (name) => `
      <div class="flex items-center gap-2 rounded-xl bg-slate-50 p-2.5 transition hover:bg-brand-50 sm:gap-3 sm:p-3">
        <div class="flex h-8 w-8 flex-shrink-0 items-center justify-center rounded-lg bg-brand-100 text-brand-600 sm:h-9 sm:w-9">
          <svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2.5"><circle cx="12" cy="12" r="3"/></svg>
        </div>
        <span class="text-xs font-semibold text-navy-700 sm:text-sm">${name}</span>
      </div>
    `,
            )
            .join("");
    };

    const renderGames = (games, sections) => {
        const list = document.querySelector("[data-games-list]");
        if (!list) return;
        const findSec = (name) => sections.find((s) => s.name === name);
        list.innerHTML =
            games
                .map((g) => {
                    const ls = findSec(g.local);
                    const vs = findSec(g.visit);
                    const live = (g.status || "").includes("En vivo");
                    return `
        <article data-game-id="${g.id}" class="overflow-hidden rounded-2xl bg-white p-5 shadow-sm ring-1 ring-slate-200 transition hover:shadow-lg sm:p-6 ${live ? "ring-2 ring-brand-400" : ""}">
          <div class="flex flex-wrap items-center justify-between gap-2 border-b border-slate-100 pb-3">
            <span class="rounded-full bg-slate-100 px-3 py-1 text-xs font-bold uppercase text-slate-600">${g.sport || ""}</span>
            <span class="rounded-full px-3 py-1 text-xs font-bold ${live ? "bg-brand-500 text-white" : "bg-slate-100 text-slate-600"}">${g.status || ""}</span>
          </div>
          <div class="mt-4 grid grid-cols-3 items-center gap-2 sm:gap-3">
            <div class="text-center">
              <div class="mx-auto mb-2 flex h-12 w-12 items-center justify-center rounded-full ${ls?.color || "bg-slate-500"} text-lg font-extrabold text-white sm:h-14 sm:w-14 sm:text-xl">${ls?.initial ?? (g.local || "?")[8] ?? "?"}</div>
              <p class="text-xs font-bold text-navy-700 sm:text-sm">${g.local || ""}</p>
            </div>
            <div class="text-center">
              <p class="text-3xl font-black text-navy-700 tabular-nums sm:text-5xl">${g.localScore} <span class="text-xl text-slate-400 sm:text-2xl">-</span> ${g.visitScore}</p>
              <p class="mt-1 text-[10px] text-slate-500 sm:text-xs">${g.date || ""}</p>
            </div>
            <div class="text-center">
              <div class="mx-auto mb-2 flex h-12 w-12 items-center justify-center rounded-full ${vs?.color || "bg-slate-500"} text-lg font-extrabold text-white sm:h-14 sm:w-14 sm:text-xl">${vs?.initial ?? (g.visit || "?")[8] ?? "?"}</div>
              <p class="text-xs font-bold text-navy-700 sm:text-sm">${g.visit || ""}</p>
            </div>
          </div>
        </article>
      `;
                })
                .join("") ||
            '<p class="col-span-full text-center text-sm text-slate-500">No hay partidos registrados.</p>';
    };

    const renderStandings = (games, sections) => {
        const tbody = document.querySelector("[data-standings-body]");
        if (!tbody) return;
        const stats = calculateStandings(games, sections);
        tbody.innerHTML = stats
            .map(
                (x, i) => `
      <tr data-standings-row="${x.id}" class="${i === 0 ? "bg-brand-50" : ""}">
        <td class="px-4 py-3 font-bold text-navy-700 sm:px-6">${i + 1}</td>
        <td class="px-4 py-3 sm:px-6">
          <div class="flex items-center gap-2 sm:gap-3">
            <span class="flex h-7 w-7 flex-shrink-0 items-center justify-center rounded-full ${x.color} text-xs font-extrabold text-white sm:h-8 sm:w-8">${x.initial}</span>
            <span class="${i === 0 ? "font-extrabold text-brand-600" : "font-semibold text-slate-700"}">${x.name}</span>
          </div>
        </td>
        <td class="px-4 py-3 text-center tabular-nums sm:px-6">${x.pj}</td>
        <td class="px-4 py-3 text-center tabular-nums sm:px-6">${x.g}</td>
        <td class="px-4 py-3 text-center tabular-nums sm:px-6">${x.e}</td>
        <td class="px-4 py-3 text-center tabular-nums sm:px-6">${x.p}</td>
        <td class="px-4 py-3 text-center font-bold tabular-nums sm:px-6 ${i === 0 ? "text-brand-600" : "text-navy-700"}">${x.pts}</td>
      </tr>
    `,
            )
            .join("");
    };

    const renderStats = (stats) => {
        document.querySelectorAll("[data-stat]").forEach((el) => {
            const key = el.getAttribute("data-stat");
            if (stats && stats[key] != null) el.textContent = stats[key];
        });
    };

    const renderLiveCount = (games) => {
        const el = document.querySelector("[data-live-count]");
        if (!el) return;
        const live = games.filter((g) =>
            (g.status || "").includes("En vivo"),
        ).length;
        el.textContent = `${live} partido${live === 1 ? "" : "s"} en vivo`;
    };

    const apply = async () => {
        const result = await load(DEFAULT_DATA_COPY);
        renderStats(result.heroStats);
        renderSections(result.sections);
        renderSports(result.sports);
        renderGames(result.games, result.sections);
        renderStandings(result.games, result.sections);
        renderLiveCount(result.games);
    };

    apply();
    onDataChanged(apply);
    })();
</script>
